cmake_minimum_required(VERSION 3.24)
project(budgetcc VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(BUDGETCC_COVERAGE "Instrument for gcov/gcovr" OFF)
option(BUDGETCC_LTO "Link-time optimisation (Phase 2, E6)" OFF)
option(BUDGETCC_WERROR "Treat warnings as errors" ON)

# ---------------------------------------------------------------- dependencies
include(FetchContent)
find_package(fmt CONFIG QUIET)
if(NOT fmt_FOUND)
  FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 11.0.2 GIT_SHALLOW ON)
  FetchContent_MakeAvailable(fmt)
endif()
find_package(CLI11 CONFIG QUIET)
if(NOT CLI11_FOUND)
  FetchContent_Declare(CLI11 GIT_REPOSITORY https://github.com/CLIUtils/CLI11.git GIT_TAG v2.4.2 GIT_SHALLOW ON)
  FetchContent_MakeAvailable(CLI11)
endif()
find_package(httplib CONFIG QUIET)
if(NOT httplib_FOUND)
  FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.1
                       GIT_SHALLOW ON SOURCE_SUBDIR header_only_no_cmake)
  FetchContent_MakeAvailable(httplib)
  add_library(httplib::httplib INTERFACE IMPORTED)
  target_include_directories(httplib::httplib INTERFACE ${httplib_SOURCE_DIR})
endif()
find_package(GTest CONFIG QUIET)
if(NOT GTest_FOUND)
  set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.15.2 GIT_SHALLOW ON)
  FetchContent_MakeAvailable(googletest)
endif()
find_package(Threads REQUIRED)

# ------------------------------------------------------------------- flags
add_library(bcc_options INTERFACE)
target_compile_features(bcc_options INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(bcc_options INTERFACE /W4 /permissive- /utf-8 /Zc:__cplusplus /fp:precise)
  if(BUDGETCC_WERROR)
    target_compile_options(bcc_options INTERFACE /WX)
  endif()
  target_compile_definitions(bcc_options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS)
  if(BUDGETCC_LTO)
    target_compile_options(bcc_options INTERFACE /GL)
    target_link_options(bcc_options INTERFACE /LTCG)
  endif()
else()
  target_compile_options(bcc_options INTERFACE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(BUDGETCC_WERROR)
    target_compile_options(bcc_options INTERFACE -Werror)
  endif()
  if(BUDGETCC_LTO)
    target_compile_options(bcc_options INTERFACE -O2 -flto)
    target_link_options(bcc_options INTERFACE -flto)
  endif()
  if(BUDGETCC_COVERAGE)
    target_compile_options(bcc_options INTERFACE --coverage -O0 -g)
    target_link_options(bcc_options INTERFACE --coverage)
  endif()
endif()
if(MINGW)
  # Run independently of whichever libstdc++/winpthread DLLs happen to be first on PATH.
  target_link_options(bcc_options INTERFACE -static -Wl,--whole-archive -lpthread -Wl,--no-whole-archive)
endif()
if(WIN32)
  target_compile_definitions(bcc_options INTERFACE _WIN32_WINNT=0x0A00)
  target_link_libraries(bcc_options INTERFACE ws2_32)
endif()
target_include_directories(bcc_options INTERFACE ${CMAKE_SOURCE_DIR}/src)
string(TOUPPER "${CMAKE_BUILD_TYPE}" _bcc_bt)
set(BUDGETCC_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${_bcc_bt}}")
if(BUDGETCC_LTO)
  string(APPEND BUDGETCC_FLAGS " LTO")
endif()
string(STRIP "${BUDGETCC_FLAGS}" BUDGETCC_FLAGS)
target_compile_definitions(bcc_options INTERFACE
  BUDGETCC_BUILD_TYPE="$<CONFIG>"
  BUDGETCC_GIT_SHA="${BUDGETCC_GIT_SHA}"
  BUDGETCC_FLAGS="${BUDGETCC_FLAGS}"
  BUDGETCC_SOURCE_DIR="${CMAKE_SOURCE_DIR}")

# --------------------------------------------------------- module libraries
function(bcc_module name)
  file(GLOB_RECURSE srcs CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/src/${name}/*.cpp)
  if(srcs)
    add_library(bcc_${name} STATIC ${srcs})
    target_link_libraries(bcc_${name} PUBLIC bcc_options ${ARGN})
  else()
    add_library(bcc_${name} INTERFACE)
    target_link_libraries(bcc_${name} INTERFACE bcc_options ${ARGN})
  endif()
endfunction()

bcc_module(core fmt::fmt Threads::Threads)
bcc_module(link bcc_core)
bcc_module(cc bcc_core)
bcc_module(transport bcc_core bcc_cc bcc_link)
bcc_module(telemetry bcc_core fmt::fmt httplib::httplib)
bcc_module(harness bcc_core bcc_link bcc_cc bcc_transport bcc_telemetry CLI11::CLI11)

set(BCC_ALL bcc_harness bcc_telemetry bcc_transport bcc_cc bcc_link bcc_core)

# --------------------------------------------------------------- executables
add_executable(budgetcc src/main.cpp)
target_link_libraries(budgetcc PRIVATE ${BCC_ALL} CLI11::CLI11)

# Benchmark logic is a library so tests/bench can exercise it; the operator-new counter lives only in
# the bench executable and the test executable, never in the budgetcc binary.
add_library(bcc_bench STATIC bench/bench_core.cpp bench/alloc_counter.cpp)
target_include_directories(bcc_bench PUBLIC ${CMAKE_SOURCE_DIR}/bench)
target_link_libraries(bcc_bench PUBLIC ${BCC_ALL})

add_executable(budgetcc_bench bench/bench_main.cpp)
target_link_libraries(budgetcc_bench PRIVATE bcc_bench CLI11::CLI11)

enable_testing()
file(GLOB_RECURSE TEST_SRCS CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/tests/*.cpp)
if(TEST_SRCS)
  add_executable(budgetcc_tests ${TEST_SRCS})
  target_include_directories(budgetcc_tests PRIVATE ${CMAKE_SOURCE_DIR}/tests ${CMAKE_SOURCE_DIR})
  target_link_libraries(budgetcc_tests PRIVATE bcc_bench ${BCC_ALL} GTest::gtest GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(budgetcc_tests DISCOVERY_TIMEOUT 60 PROPERTIES TIMEOUT 120)
endif()
